Lesson — Structures in C: Examples

Learning Outcome(s): LO5 — Develop code using C structures, typedefs, and dynamic allocation safely.

Assessment alignment: Final Exam (Structs: layout, init, pass-by-value vs pointer, heap management).

Basic structs and typedef (header)

In practice, define structs in a header so multiple source files can include them.

structs.h

#ifndef STRUCTS_H
#define STRUCTS_H

#define NAME_SIZE        5

// How to define a struct without a typedef.
// We rarely create structs this way - too cumbersome.
struct BirthDate
{
    // Attributes
    char cName[NAME_SIZE];
    char cGender;
    int  day;
    int  month;
    int  year;
};

// Example of declaring a struct using a typedef
typedef struct
{
    // Attributes
    char cName[NAME_SIZE];
    char cGender;
    int  day;
    int  month;
    int  year;
} BirthDateTD;

#endif // !STRUCTS_H

Size and alignment

Alignment can affect struct size. On typical ABIs, members are aligned (e.g., 4‑byte boundaries for int). The order of members can change padding → which changes sizeof(struct).

program.c (check sizes)

#include <stdio.h>
#include <stdlib.h>
#include "structs.h"

int main(void)
{
    printf("Size of struct BirthDate is: %zu\n", sizeof(struct BirthDate));
    printf("Size of BirthDateTD is: %zu\n", sizeof(BirthDateTD));
    return EXIT_SUCCESS;
}

With NAME_SIZE=5 and default packing (common x86/x64 ABIs), you may see 20 bytes due to padding. If you move cGender last, you may see 24 bytes (extra padding).

Change packing with #pragma pack (MSVC)

To demonstrate forced 1‑byte packing (implementation-specific; MSVC), wrap the typedef with:

#pragma pack(push)     // save current packing
#pragma pack(1)        // set byte alignment to 1

typedef struct
{
    char cName[NAME_SIZE];
    char cGender;
    int  day;
    int  month;
    int  year;
} BirthDateTD;

#pragma pack(pop)      // restore default packing

Now sizeof(BirthDateTD) should print as 18 (5 + 1 + 4 + 4 + 4) regardless of field order.

Basic struct usage

Initialize and display (stack-allocated)

// structs.c
#include <stdio.h>
#include "structs.h"

void DisplayBirthDateBasic(struct BirthDate bd)
{
    printf("\nName is %s\n", bd.cName);
    printf("Gender is %c\n", bd.cGender);
    printf("Date is %d %d %d\n", bd.day, bd.month, bd.year);
}
// program.c (continued)
void testStructureBasicFormat(void)
{
    // Creates a struct instance on the stack
    struct BirthDate bd1;

    // Uninitialized members contain indeterminate data (garbage).
    bd1.day = 17;
    DisplayBirthDateBasic(bd1);

    // Correct full initialization in declared order:
    struct BirthDate bd2 = { "Ada", 'f', 21, 3, 1987 };
    DisplayBirthDateBasic(bd2);

    // Wrong order / wrong count: shows why order matters and C won't validate:
    // struct BirthDate bad = { "Ada", 21, 3, 1987 };        // wrong types/order
    // struct BirthDate alsoBad = { "Ada", 21, 3, 1987, 'f' }; // extra item

    // Update an attribute
    bd2.month = 4;
    DisplayBirthDateBasic(bd2);

    // Direct memory manipulation (demonstration only; not good practice):
    // day is an int; &bd2.day + 2 moves past month (int) to year (int)
    *(&bd2.day + 2) = 1999;
    DisplayBirthDateBasic(bd2);
}

Typedef style (cleaner declarations)

Display and use typedef’d struct

// structs.c
void DisplayBirthDateTypedef(BirthDateTD bd)
{
    printf("\nName is %s\n", bd.cName);
    printf("Gender is %c\n", bd.cGender);
    printf("Date is %d %d %d\n", bd.day, bd.month, bd.year);
}
// program.c
void testStructureDeclarationsWithTypedef(void)
{
    BirthDateTD bd = { "Mike", 'm', 31, 12, 1980 };
    DisplayBirthDateTypedef(bd);

    // Struct assignment copies bytes (value semantics)
    BirthDateTD bd1 = bd;   // copy
    bd1.day = 1;            // does not affect bd

    DisplayBirthDateTypedef(bd);
    DisplayBirthDateTypedef(bd1);
}

Pass-by-value vs pointer (by-ref)

// structs.c
void AlterBirthDate(BirthDateTD bd)
{
    printf("\nSize of passed in BirthDateTD is %zu\n", sizeof(bd)); // e.g., 18
    bd.year = 2001;                        // modifies the local copy only
    DisplayBirthDateTypedef(bd);
}

void AlterBirthDateByRef(BirthDateTD* bdPtr)
{
    printf("\nSize of passed in bdPtr: %zu\n", sizeof(bdPtr)); // pointer size
    // (*bdPtr).year = 2023;  // clumsy
    bdPtr->year = 2023;       // preferred: arrow operator
}
// program.c
void demoPassSemantics(void)
{
    BirthDateTD bd = { "Mike", 'm', 31, 12, 1980 };

    AlterBirthDate(bd);       // changes only inside the function
    DisplayBirthDateTypedef(bd);

    AlterBirthDateByRef(&bd); // changes caller's instance
    DisplayBirthDateTypedef(bd);
}

Dynamic allocation of structs

Incorrect: using a pointer without allocating memory

void testDynamicallyAllocatedStructs_bad(void)
{
    BirthDateTD* bdPtr = NULL;

    // Crashes: no memory allocated for the struct yet
    // bdPtr->day = 12;
}

Correct: allocate, use, then free

#include <stdlib.h>
#include <string.h>

void testDynamicallyAllocatedStructs(void)
{
    BirthDateTD* bdPtr = (BirthDateTD*)malloc(sizeof(BirthDateTD));
    if (bdPtr != NULL)
    {
        bdPtr->day = 13;
        strcpy_s(bdPtr->cName, NAME_SIZE, "Mike"); // include <string.h>

        // Display function expects a struct instance, not pointer
        DisplayBirthDateTypedef(*bdPtr);

        // Always free heap memory you own
        free(bdPtr);
        bdPtr = NULL;
    }
}

Structs with dynamically allocated attributes

student (name and marks allocated dynamically)

typedef struct
{
    // Some of the attributes are dynamically allocated
    char* cNamePtr;
    int   iStudentNum;

    // Array of marks
    int*  iMarksPtr;
    // Number of marks in the marks array
    int   iMarksCount;
} student;

On x64, pointer members align to 8 bytes; on x86 to 4 bytes, so sizeof(student) varies by platform.

Test driver

void testStudent(void)
{
    int iMarksArray[] = { 95, 100, 85 };

    student* sPtr = CreateStudent("Mike Grzesina", 123456789,
                                  iMarksArray,
                                  (int)(sizeof(iMarksArray) / sizeof(int)));

    DisplayStudent(*sPtr);
    FreeStudent(sPtr);
    sPtr = NULL;
}

Create / Display / Free

#include <stdlib.h>
#include <string.h>

// Create and initialize a student (like “new” + constructor)
student* CreateStudent(char* cNamePtr, int iNum, int* iMarksPtr, int iMarkCount)
{
    student* sPtr = (student*)malloc(sizeof(student));

    if (sPtr != NULL)
    {
        // name
        sPtr->cNamePtr = (char*)malloc((strlen(cNamePtr) + 1) * sizeof(char));
        strcpy(sPtr->cNamePtr, cNamePtr);

        // id
        sPtr->iStudentNum = iNum;

        // marks
        sPtr->iMarksPtr = (int*)malloc(iMarkCount * sizeof(int));
        memcpy(sPtr->iMarksPtr, iMarksPtr, iMarkCount * sizeof(int));
        sPtr->iMarksCount = iMarkCount;
    }
    return sPtr;
}

// Display by value (copy) to avoid accidental modification
void DisplayStudent(student s)
{
    printf("\nName: %s\n", s.cNamePtr);
    printf("Student number: %d\n", s.iStudentNum);
    printf("Marks: ");
    for (int i = 0; i < s.iMarksCount; i++)
    {
        printf("%d ", s.iMarksPtr[i]);
    }
    printf("\n");
}

// Free nested allocations first, then the struct
void FreeStudent(student* sPtr)
{
    free(sPtr->cNamePtr);  sPtr->cNamePtr = NULL;
    free(sPtr->iMarksPtr); sPtr->iMarksPtr = NULL;

    free(sPtr); // design decision: free only if dynamically allocated
}

Pointers to pointers to structs (e.g., classroom with student list)

classroom with array of student*

typedef struct
{
    // A dynamically allocated array of student pointers
    student** sPtrPtr;
    // How many students are in the classroom
    unsigned short sNumStudents;
} classroom;

Get one student from user

#define MAX_NAME_SIZE 256

student* GetStudent(void)
{
    char cName[MAX_NAME_SIZE];
    int  iMarks[2];
    int  iStudentNum;

    // Name
    printf("Enter the student name: ");
    gets_s(cName, MAX_NAME_SIZE);

    // Number
    printf("Enter the student number: ");
    scanf("%d", &iStudentNum);

    // Two marks
    printf("Enter two marks separated by a space: ");
    scanf("%d %d", iMarks, iMarks + 1);
    getc(stdin); // clear newline

    return CreateStudent(cName, iStudentNum, iMarks, 2);
}

Populate, display, and free classroom

classroom PopulateClassroom(void)
{
    classroom c;

    printf("How many students are in the classroom? ");
    scanf("%hu", &c.sNumStudents);
    getc(stdin);

    c.sPtrPtr = (student**)malloc(c.sNumStudents * sizeof(student*));

    for (int i = 0; i < c.sNumStudents; i++)
    {
        c.sPtrPtr[i] = GetStudent();
    }
    return c;
}

void DisplayClassroom(classroom c)
{
    printf("\nDisplaying the classroom:\n");
    for (int i = 0; i < c.sNumStudents; i++)
    {
        DisplayStudent(*(c.sPtrPtr[i])); // pass by value
    }
}

void FreeClassroom(classroom* classPtr)
{
    // Free each student (which frees nested allocations)
    for (int i = 0; i < classPtr->sNumStudents; i++)
    {
        FreeStudent(classPtr->sPtrPtr[i]);
        classPtr->sPtrPtr[i] = NULL;
    }

    // Free the array of pointers
    free(classPtr->sPtrPtr);
    classPtr->sPtrPtr = NULL;

    // We did not heap-allocate 'classroom' itself in this design; do not free it.
}
void testClassroom(void)
{
    classroom c = PopulateClassroom();
    DisplayClassroom(c);
    FreeClassroom(&c);
}

Notes & Tips

  • Uninitialized struct members contain indeterminate values (garbage). Always initialize before use.
  • Struct assignment copies all bytes (value semantics). Passing by value copies; passing a pointer does not.
  • When freeing nested dynamic fields, free children first, then the parent struct.
  • #pragma pack is compiler/ABI-specific; use for demonstration/testing, not typically in production portability scenarios.
