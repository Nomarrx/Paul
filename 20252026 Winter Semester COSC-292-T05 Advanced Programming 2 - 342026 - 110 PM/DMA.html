Lesson — Dynamic Memory Allocation

Learning Outcome(s): LO5 — Develop code to reference and manipulate static and dynamic memory with pointers.

Assessment alignment: Midterm 2 (Memory & DMA).

Dynamic Memory Allocation (DMA)

Stack memory lives only as long as a function’s stack frame; you cannot safely “return” stack memory to the caller. The solution is to allocate on the heap (via malloc and friends), which persists until you explicitly free it.

In C:

  • malloc(N) returns a void* to N bytes of uninitialized memory (or NULL on failure).
  • free(ptr) releases a previously allocated block.
  • realloc(ptr, newSize) resizes a block (may move and free the old one).
  • Headers: <stdlib.h> (and/or <malloc.h>).

Utility: print an array of ints

Project setup

#include <stdlib.h>
#include <stdio.h>
#include "dma.h"

int main(void)
{
    int iArray[] = { 10, 20, 30 };
    printInts(iArray, (int)(sizeof(iArray) / sizeof(int)));
    return EXIT_SUCCESS;
}

dma.h

#ifndef DMA_H
#define DMA_H

// Function to print an array of ints
void printInts(int* iArrayPtr, int iSize);

// DMA helpers we will add later
int* getInts(int iNumInts);
int* addInt(int* iArrayPtr, int* iSizePtr);
int* addIntRealloc(int* iArrayPtr, int* iSizePtr);

#endif // !DMA_H

dma.c (printer)

#include <stdio.h>
#include "dma.h"

// Prints a comma-separated list. Assumes iSize >= 1.
void printInts(int* iArrayPtr, int iSize)
{
    printf("The integers are: ");
    for (int i = 0; i < iSize - 1; i++)
    {
        printf("%d, ", *iArrayPtr);
        iArrayPtr++;
    }
    printf("%d\n", *iArrayPtr);
}

Output: The integers are: 10, 20, 30

Exercise 1 — Allocate ints on the heap with malloc

Goal

Allocate an array of int on the heap, return it, and print it. Then free it properly.

Test driver

void testIntDMA(void)
{
    int iSize = 3;
    int* iBufferPtr = getInts(iSize);

    printInts(iBufferPtr, iSize);

    free(iBufferPtr);   // Always free heap memory you own
    iBufferPtr = NULL;  // Safety: null the pointer after free
}

Implementation — initial version (uninitialized memory)

#include <stdlib.h>   // malloc
#include <stdio.h>
#include "dma.h"

// Return an array of ints that are dynamically allocated
int* getInts(int iNumInts)
{
    int* iBufferPtr = (int*)malloc(iNumInts * sizeof(int));
    return iBufferPtr; // may be NULL on failure; memory is uninitialized
}

What to notice

  • Uninitialized: malloc memory contains garbage. Don’t assume zeros.
  • NULL check: Always check return value against NULL before use.
  • free + NULL: After free, set the pointer to NULL to avoid use‑after‑free.

Improvement A — initialize with memset

#include <string.h>   // memset

int* getInts(int iNumInts)
{
    int* iBufferPtr = (int*)malloc(iNumInts * sizeof(int));
    if (iBufferPtr != NULL)
    {
        memset(iBufferPtr, 0, iNumInts * sizeof(int));
    }
    return iBufferPtr;
}

Improvement B — input values into the array

int* getInts(int iNumInts)
{
    int* iBufferPtr = (int*)malloc(iNumInts * sizeof(int));

    if (iBufferPtr != NULL)
    {
        for (int i = 0; i < iNumInts; i++)
        {
            printf("Enter integer %d: ", i + 1);
            scanf_s("%d", iBufferPtr + i);
        }
    }
    return iBufferPtr;
}

Important: C does not perform bounds checks. Writing past the end of your allocation is undefined behavior.

Exercise 2 — Grow the array: copy + free vs realloc

Task

Add one more int to the end of the heap array and update its size. Do it first by allocating a new buffer and copying; then do it with realloc.

Driver

void testIntDMA(void)
{
    int iSize = 3;
    int* iBufferPtr = getInts(iSize);

    printInts(iBufferPtr, iSize);

    // Add another int (copy + free old)
    iBufferPtr = addInt(iBufferPtr, &iSize);
    printInts(iBufferPtr, iSize);

    // Add another int (realloc)
    iBufferPtr = addIntRealloc(iBufferPtr, &iSize);
    printInts(iBufferPtr, iSize);

    free(iBufferPtr);
    iBufferPtr = NULL;
}

Solution A — allocate new + memcpy + free old

#include <string.h> // memcpy

// Add an int to the end of a dynamic array
// Purpose: grow by one, prompt for the new value, return the new buffer.
int* addInt(int* iArrayPtr, int* iSizePtr)
{
    int newSize = *iSizePtr + 1;

    int* iNewPtr = (int*)malloc(newSize * sizeof(int));
    if (iNewPtr == NULL)
    {
        // Allocation failed; return original pointer unchanged
        return iArrayPtr;
    }

    memcpy(iNewPtr, iArrayPtr, (*iSizePtr) * sizeof(int));

    printf("Enter another integer: ");
    scanf_s("%d", iNewPtr + *iSizePtr);

    // Increment the size
    (*iSizePtr)++;

    // Free the original buffer to avoid a memory leak
    free(iArrayPtr);
    iArrayPtr = NULL;

    return iNewPtr;
}

Solution B — use realloc (C89)

// Add an int to the end of a dynamic array using realloc
int* addIntRealloc(int* iArrayPtr, int* iSizePtr)
{
    int newSize = *iSizePtr + 1;

    // Realloc may move the block and free the old one internally.
    int* iNewPtr = (int*)realloc(iArrayPtr, newSize * sizeof(int));
    if (iNewPtr != NULL)
    {
        printf("Enter another integer: ");
        scanf_s("%d", iNewPtr + *iSizePtr);

        (*iSizePtr)++;
        return iNewPtr;
    }
    else
    {
        // On failure, original pointer is still valid; return it unchanged
        return iArrayPtr;
    }
}

 

Detecting memory leaks (Visual Studio)

Windows/VS can show leaks by mapping allocated blocks and dumping them on exit in debug runs.

At the top of your program.c

// for detecting memory leaks
#define _CRTDBG_MAP_ALLOC
#include <crtdbg.h>

At the end of main (before return)

_CrtDumpMemoryLeaks();

Comment out your free calls to see the debugger’s leak report (addresses and sizes will vary).

Exercise 3 — Return dynamically allocated strings of the minimum size

Task

Write getStringDynamic(prompt) that:

  1. Temporarily reads into a buffer,
  2. Allocates the exact required size (length + 1),
  3. Copies into the right‑sized buffer and returns it,
  4. Frees the temporary buffer.

Test driver

void testDynamicString(void)
{
    char* cFirstPtr = NULL;
    char* cLastPtr  = NULL;

    cFirstPtr = getStringDynamic("Enter first name: ");
    cLastPtr  = getStringDynamic("Enter last name: ");

    if (cFirstPtr)
    {
        printf("The first name is %s\n", cFirstPtr);
        free(cFirstPtr);
        cFirstPtr = NULL;
    }
    if (cLastPtr)
    {
        printf("The last name is %s\n", cLastPtr);
        free(cLastPtr);
        cLastPtr = NULL;
    }
}

Wrong approach (returns stack memory — don’t do this)

#define MAX_STRING_SIZE 256

char* getStringDynamic(const char* cPrompt)
{
    char cReturn[MAX_STRING_SIZE]; // stack!

    printf("%s", cPrompt);
    gets_s(cReturn, MAX_STRING_SIZE);

    return cReturn; // BUG: address of local (stack) memory
}

Correct approach (two allocations: temporary + exact size)

#define _CRT_SECURE_NO_WARNINGS
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

#define MAX_STRING_SIZE 256

// Return a pointer to a dynamically allocated string or NULL if it fails
char* getStringDynamic(const char* cPrompt)
{
    char* cTempPtr   = (char*)malloc(MAX_STRING_SIZE * sizeof(char));
    char* cReturnPtr = NULL;

    if (cTempPtr)
    {
        printf("%s", cPrompt);
        gets_s(cTempPtr, MAX_STRING_SIZE);

        int iSizeDestination = (int)strlen(cTempPtr) + 1;
        cReturnPtr = (char*)malloc(iSizeDestination * sizeof(char));
        if (cReturnPtr)
        {
            strcpy_s(cReturnPtr, iSizeDestination, cTempPtr);
        }

        free(cTempPtr);
        cTempPtr = NULL;
    }
    return cReturnPtr;
}

Remember: The caller owns the returned pointer and must free it.

Exercise 4 — Packed record (name + SIN) in one buffer

Learning goals

  • Allocate a buffer containing a fixed-length char region followed by an int.
  • Navigate the buffer using pointer arithmetic and casts.

Task

  1. Write getNameAndSIN() to allocate MAX_NAME_SIZE bytes for the name + sizeof(int) for the SIN.
  2. Read the name (using fgets) and trim newline if present.
  3. Read the SIN as an int (store immediately after the name region).
  4. Write printNameAndSIN(buffer) to display both.

Prototypes

Solution — allocate, read, and return

#include <stdlib.h>
#include <stdio.h>
#include <string.h>

char* getNameAndSIN(void)
{
    char* cNameAndSINPtr = (char*)malloc(MAX_NAME_SIZE * sizeof(char) + sizeof(int));
    if (cNameAndSINPtr)
    {
        int iLength = 0;

        // Read name into first MAX_NAME_SIZE bytes
        printf("Enter the name: ");
        fgets(cNameAndSINPtr, MAX_NAME_SIZE, stdin);

        // Trim newline if present
        iLength = (int)strlen(cNameAndSINPtr);
        if (iLength > 0 && cNameAndSINPtr[iLength - 1] == '\n')
        {
            cNameAndSINPtr[iLength - 1] = '\0';
        }

        // Read SIN into the int immediately after the name region
        printf("Enter the SIN#: ");
        scanf_s("%d", (int*)(cNameAndSINPtr + MAX_NAME_SIZE));
    }
    return cNameAndSINPtr;
}

Solution — print function

void printNameAndSIN(char* cBuffer)
{
    int* iSINPtr = (int*)(cBuffer + MAX_NAME_SIZE);
    printf("The name is %s\n", cBuffer);
    printf("The SIN# is %d\n", *iSINPtr);
}

 

Driver

void testNameAndSIN(void)
{
    char* cBuffer = getNameAndSIN();
    if (cBuffer)
    {
        printNameAndSIN(cBuffer);
        free(cBuffer);
        cBuffer = NULL;
    }
}

Solution

#include <string.h>
#include <stdlib.h>

// Returns a dynamically allocated string (caller must free)
char* concatenateStrings(const char* cStringPtr1, const char* cStringPtr2)
{
    int iLengthFirst  = (int)strlen(cStringPtr1);
    int iLengthSecond = (int)strlen(cStringPtr2);

    char* cReturnString = (char*)malloc(iLengthFirst + iLengthSecond + 1);
    if (cReturnString)
    {
        // Initialize destination as empty
        cReturnString[0] = '\0';

        // Copy s1
        strcpy_s(cReturnString, iLengthFirst + 1, cStringPtr1);

        // Copy s2 directly to the end of s1
        strcpy_s(cReturnString + iLengthFirst, iLengthSecond + 1, cStringPtr2);
        // Alternatively: strcat_s(cReturnString, iLengthFirst + iLengthSecond + 1, cStringPtr2);
    }
    return cReturnString;
}
